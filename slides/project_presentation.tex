\begin{lstlisting}[basicstyle=\ttfamily\large]
for name in name_list:
  for token in name:
    embedding(token)
    for layer in layers:
      attention(token)
      mlp(token)
    calc_loss(token)
sum_loss(tokens)
\end{lstlisting}
        \end{column}
    \end{columns}
\end{frame}


\begin{frame}{Serial Analysis}
    \textbf{Big-O: } \\
    \begin{itemize}
        \item T = sequence length, E = embedding size, L = layer, N = names
        \item $O(N \cdot L \cdot (C_{\text{attn}} + C_{\text{mlp}}))$ \\ 
        \item $O(C_{\text{attn}})$ $ = T^2 \cdot E$ \\
        \item $O(C_{\text{mlp}})$ $ = T \cdot E^{2}$\\ 
        \item Since names are mostly constant size it becomes $\approx O(N \cdot L \cdot E^2)$
    \end{itemize}

    \vspace{0.1cm}
    \large
    \textbf{Memory vs CPU Bound: }
    \begin{itemize}
        \item Total Time: 6.41 s, CPU Time: 6.19
        \item CPU utilization $\approx \frac{6.19}{6.41} \approx 96.6\% \Rightarrow$ CPU-bound
    \end{itemize}
    
    \vspace{0.1cm}
    \textbf{Static Analysis:}
    \begin{itemize}
        \item \textbf{99.4\%} of work spent doing matrix mult, (0.4\% for rmnsnorm, SM, emb)
        \item \textbf{71.1\%} for MLP \textbf{28.3\%} for ATTN  
        \item By Amdahl’s Law, with $p = 0.998$, the theoretical maximum speedup is $\frac{1}{1 - p} \approx 500\times$
    \end{itemize}
\end{frame}

    \vspace{0.05cm}
    {\large\texttt{name\_list: [[BOS, E, L, L, A, BOS], [name\_B], [name\_C]]}\par}
    \vspace{0.2cm}
    {\large\texttt{BOS = Beginning of Sequence}\par}

    \vspace{0.3cm}
    \begin{columns}[T,onlytextwidth]

        % -------- SERIAL --------
        \begin{column}{0.49\textwidth}
        \textbf{Serial} 
\begin{lstlisting}
for name in name_list:
    for token in name:
        embedding(token)
        for layer in layers:
            attention(token)
            mlp(token)
        calc_loss(token)
sum_loss(tokens)
\end{lstlisting}
        \end{column}

        % -------- PARALLEL --------
        \begin{column}{0.49\textwidth}
        \textbf{Parallel (per batch)}
\begin{lstlisting}
calculate_embeddings(name_list)

for layer in layers:
    calculate_attention_scores(name_list)
    calculate_next_hidden_state(name_list)

calc_loss(name_list) 
\end{lstlisting}
        \end{column}

    \end{columns}
\end{frame}

\begin{frame}{Parallelization Strategy}
    \vspace{0.2cm}
    \begin{itemize}
        \setlength\itemsep{0.18cm}
        \item \textbf{Batching}
            \begin{itemize}
                \item For each sequence launch a transformer $\rightarrow$ run transformer on all sequences at once
            \end{itemize}
        \item Store all of our information in a flattened array
            \begin{itemize}
                \item Allows assignment of work easily by batch to threads
            \end{itemize}
        \item Store a ``workspace'' on the GPU that holds weights, hidden states, etc.
        \item Large portions of the forward pass are parallelizable
            \begin{itemize}
                \item \textbf{Embedding calculations}
                \item \textbf{RMS normalization and linear calculations}
                \item \textbf{Attention}
            \end{itemize}
        \item \textbf{Layers must be performed sequentially}
    \end{itemize}
\end{frame}
